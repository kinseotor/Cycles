const gpuTexture = device.createTexture({
  label: "MyTextureArray",
  size: [textureSize.width, textureSize.height],
  arrayLayerCount: layerCount, // ✅ 数组层数，核心
  mipLevelCount: 1,
  sampleCount: 1,
  dimension: "2d", // 2d数组纹理，固定写2d！不是3d
  format: "rgba8unorm",
  usage: GPUTextureUsage.TEXTURE_BINDING | GPUTextureUsage.COPY_DST
});

const layer0Data = new Uint8Array(256*256*4);
device.queue.writeTexture(
  { texture: gpuTexture, origin: [0,0,0] }, // origin[2] = layer 0
  layer0Data,
  { bytesPerRow: 256 * 4 },
  [256,256,1] // depthOrArrayLayers:1，只写1层
);

const bindLayout = device.createBindGroupLayout({
  entries: [{
    binding: 0,
    visibility: GPUShaderStage.FRAGMENT,
    texture: { viewDimension: "2d-array" } // ✅ 标记是2d数组纹理视图
  },{
    binding: 1,
    visibility: GPUShaderStage.FRAGMENT,
    sampler: {}
  }]
});

// 重点：创建 textureView，必须设置 dimension:"2d-array"
const textureView = gpuTexture.createView({
  dimension: "2d-array"
});

const bindGroup = device.createBindGroup({
  layout: bindLayout,
  entries: [
    { binding:0, resource: textureView },
    { binding:1, resource: sampler }
  ]
});
@group(0) @binding(0) var tex_arr: texture_2d_array<f32>;
@group(0) @binding(1) var samp: sampler;

@fragment
fn fs_main(@builtin(position) pos: vec4f) -> @location(0) vec4f {
  let uv = pos.xy / vec2f(256,256);
  
  let layer0 = textureSample(tex_arr, samp, uv, 0.0); // 取第0层
  let layer1 = textureSample(tex_arr, samp, uv, 1.0); // 取第1层
  
  return mix(layer0, layer1, 0.5);
}
