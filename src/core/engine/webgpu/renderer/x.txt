        const h = Texture2D.defaultHeight; //2048
        const layer0Data = new Uint8Array(w * h * 4); 

        // 填充默认颜色，比如纯白
        for(let i=0;i< w*h*4;i+=4){

            layer0Data[i+1]=255;
            layer0Data[i+2]=255;
            layer0Data[i+3]=255;

            { texture: this.Resource.Texture.Texture2DArray, origin: [0,0,0] },
            layer0Data,
